# 1. Print your current working directory (shows where you are)
pwd

# 2. List all files and folders in the current directory
ls

# 3. Create a new practice folder
mkdir terminal-practice

# 4. Move inside the newly created folder
cd terminal-practice

# 5. Verify you successfully moved inside the folder
pwd

# 6. Create a blank file inside the folder (Mac/Linux)
# Note: For Windows PowerShell, use: New-Item practice.txt -ItemType File
touch practice.txt

# 7. List contents to confirm the file was created
ls

# 8. Navigate back out to the parent folder
cd ..

# 9. Final check of your current directory path
pwd
